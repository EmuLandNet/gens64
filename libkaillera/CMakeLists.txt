# libkaillera - Kaillera netplay client. LGPL-2.1-or-later, see LICENSE.
cmake_minimum_required(VERSION 3.13)
project(libkaillera C)

add_library(kaillera STATIC src/kc.c src/kc_net.c)
target_include_directories(kaillera PUBLIC include)
set_target_properties(kaillera PROPERTIES C_STANDARD 99)
if(WIN32)
	target_link_libraries(kaillera PUBLIC ws2_32)
else()
	target_compile_definitions(kaillera PRIVATE _POSIX_C_SOURCE=200809L)
endif()

option(KAILLERA_TEST_CLIENT "Build the console test client" OFF)
if(KAILLERA_TEST_CLIENT)
	add_executable(kc_test tools/kc_test.c)
	target_link_libraries(kc_test kaillera)
endif()

# kailleraclient.dll: the original client's interface with Windows dialogs
if(WIN32)
	if(CMAKE_SIZEOF_VOID_P EQUAL 8)
		set(KCL_DEF client/kailleraclient64.def)
	else()
		set(KCL_DEF client/kailleraclient32.def)
	endif()
	add_library(kailleraclient SHARED client/kcl.c client/kailleraclient.rc ${KCL_DEF})
	target_include_directories(kailleraclient PRIVATE client)
	target_link_libraries(kailleraclient PRIVATE kaillera comctl32 wininet ws2_32)
	set_target_properties(kailleraclient PROPERTIES PREFIX "" C_STANDARD 99)
	target_link_options(kailleraclient PRIVATE -static-libgcc -s)
endif()
